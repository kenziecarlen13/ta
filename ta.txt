## Cell 3: Mesin Pra-Pemrosesan Data MTF (Data Preprocessing Engine)
Menerapkan *feature engineering* multi-timeframe yang ketat:
1. **Zero Lookahead Bias:** Indikator tren makro harian (EMA 21 dan EMA 63 pada timeframe 1D) digeser satu bar (`.shift(1)`) sebelum digabungkan (*forward-fill*) ke dalam data per jam (1H).
2. **Sanitasi Jam Bursa Reguler (RTH):** Hanya memproses jam perdagangan reguler bursa AS (09:00 - 16:00 EST) serta mengeliminasi bar akhir pekan (Sabtu/Minggu).

# CELL 3: DATA PREPROCESSING ENGINE
# =============================================================================
def calc_ema_series(series, span):
    """Menghitung Exponential Moving Average (EMA) berbasis span."""
    return pd.Series(series).ewm(span=span, adjust=False).mean().values

def calc_atr_series(high, low, close, period=14):
    """Menghitung Average True Range (ATR) berbasis True Range Wilder/EMA."""
    high_s = pd.Series(high)
    low_s = pd.Series(low)
    close_s = pd.Series(close)
    close_prev = close_s.shift(1)
    tr = pd.concat([
        high_s - low_s,
        (high_s - close_prev).abs(),
        (low_s - close_prev).abs()
    ], axis=1).max(axis=1)
    return pd.Series(tr).ewm(span=period, adjust=False).mean().values

def load_clean_data(file_path, is_1h=False):
    """Membaca data CSV/Parquet, sanitasi jam RTH 09:00-16:00, dan eliminasi akhir pekan."""
    p = Path(file_path)
    if not p.exists():
        raise FileNotFoundError(f"Berkas data tidak ditemukan pada path: {p}")
        
    if p.suffix.lower() == '.parquet':
        df = pd.read_parquet(p)
    else:
        with open(p, 'r', encoding='utf-8') as f:
            first_line = f.readline()
            sep = '\t' if '\t' in first_line else ','
        df = pd.read_csv(p, sep=sep)
        
    df.columns = [c.lower() for c in df.columns]
    
    # Standarisasi kolom waktu
    if 'datetime' in df.columns:
        df['datetime'] = pd.to_datetime(df['datetime'])
    elif 'date' in df.columns:
        df['datetime'] = pd.to_datetime(df['date'])
        df.drop(columns=['date'], inplace=True, errors='ignore')
    elif 'time' in df.columns:
        df['datetime'] = pd.to_datetime(df['time'])
        
    df.sort_values('datetime', inplace=True)
    df.set_index('datetime', inplace=True)
    
    # Eliminasi akhir pekan (Sabtu=5, Minggu=6)
    df = df[df.index.dayofweek < 5]
    
    # Filter jam bursa reguler (Regular Trading Hours 09:00 - 16:00 EST)
    if is_1h:
        df = df[(df.index.hour >= 9) & (df.index.hour <= 16)]
        
    df['date'] = df.index.date
    return df

def get_asset_file_paths(asset):
    """Pemetaan adaptif berkas data 1D dan 1H per instrumen."""
    search_dirs = [DATA_DIR, BASE_DIR / "data_index", BASE_DIR.parent / "TugasAkhir" / "data_index"]
    for d in search_dirs:
        mapping = {
            'SPY': {'1d': d / 'spy_US500_D1.parquet', '1h': d / 'spy_US500_H1.parquet'},
            'QQQ': {'1d': d / 'Nasdaq100_1d_data.csv', '1h': d / 'Nasdaq100_1h_data.csv'},
            'DJI': {'1d': d / 'Dow_Jones_1d_data.csv', '1h': d / 'Dow_Jones_1h_data.csv'},
            'AAPL': {'1d': d / 'validasi-aapl' / 'aapl_1d.csv', '1h': d / 'validasi-aapl' / 'aapl_1h.csv'}
        }
        if mapping[asset]['1d'].exists() and mapping[asset]['1h'].exists():
            return mapping[asset]
            
    raise FileNotFoundError(f"Tidak dapat menemukan berkas data lengkap untuk instrumen: {asset}")

def preprocess_asset_data(asset, start_dt='2018-02-19', end_dt='2024-05-10 23:59:59'):
    """
    Pipeline Feature Engineering Multi-Timeframe:
    1. Menghitung indikator 1D secara utuh pada data penuh (Full Period) dengan pergeseran 1 bar (.shift(1))
       guna mencegah kebocoran data masa depan (Zero Lookahead Bias).
    2. Menyaring data 1H pada rentang waktu evaluasi seragam (2018-2024).
    3. Menggabungkan indikator 1D ke 1H berbasis tanggal.
    4. Menghitung EMA Mikro 32 dan ATR 14 pada 1H.
    """
    paths = get_asset_file_paths(asset)
    d1 = load_clean_data(paths['1d'], is_1h=False)
    h1 = load_clean_data(paths['1h'], is_1h=True)
    
    # 1. Feature Engineering Dimensi Makro 1D (Shift 1 bar untuk kausalitas)
    d1['ema_fast'] = pd.Series(calc_ema_series(d1['close'], GLOBAL_PARAMS['ema_macro_fast']), index=d1.index).shift(1)
    d1['ema_slow'] = pd.Series(calc_ema_series(d1['close'], GLOBAL_PARAMS['ema_macro_slow']), index=d1.index).shift(1)
    thresh = GLOBAL_PARAMS['macro_spread_threshold']
    
    # Aturan Makro Simetris:
    d1['macro_bullish'] = d1['ema_fast'] > (d1['ema_slow'] * (1.0 + thresh))
    d1['macro_bearish'] = d1['ema_fast'] < (d1['ema_slow'] * (1.0 - thresh))
    d1['macro_bull_alive'] = d1['ema_fast'] >= d1['ema_slow']
    d1['macro_bear_alive'] = d1['ema_fast'] <= d1['ema_slow']
    
    # 2. Sinkronisasi rentang waktu evaluasi bersama
    h1 = h1[(h1.index >= pd.to_datetime(start_dt)) & (h1.index <= pd.to_datetime(end_dt))]
    
    # 3. Penyelarasan MTF 1D ke 1H berbasis tanggal
    d1_sub = d1[['date', 'macro_bullish', 'macro_bearish', 'macro_bull_alive', 'macro_bear_alive']]
    df = pd.merge(h1.reset_index(), d1_sub, on='date', how='left')
    df.ffill(inplace=True)
    df.set_index('datetime', inplace=True)
    
    # 4. Standardisasi format kolom kapital sesuai spesifikasi Backtesting.py
    col_rename = {'open': 'Open', 'high': 'High', 'low': 'Low', 'close': 'Close', 'volume': 'Volume'}
    df.rename(columns=col_rename, inplace=True)
    
    # 5. Feature Engineering Dimensi Mikro 1H
    df['EMA_micro'] = calc_ema_series(df['Close'], GLOBAL_PARAMS['ema_micro'])
    df['ATR'] = calc_atr_series(df['High'], df['Low'], df['Close'], GLOBAL_PARAMS['atr_period'])
    df.dropna(inplace=True)
    
    return df

## Panduan Detail Alur Entry dan Exit Strategi (Cell 4: SymmetricMTFStrategy)
Seluruh aturan perdagangan (*rule-based quantitative rules*) yang dieksekusi oleh kelas strategi `SymmetricMTFStrategy` dirancang secara simetris dengan pembagian tahapan berikut:

---

### 1. Masuk Ketika Kondisinya Belum Apa (Prasyarat & Status Portofolio / Pre-Conditions)
Sebelum indikator teknikal dievaluasi untuk mengeksekusi order baru, algoritma memeriksa batasan operasional akun. Sistem **hanya boleh masuk jika seluruh kondisi berikut terpenuhi**:
1. **Portofolio Belum Memiliki Posisi Aktif (*Flat State*):**
   * Kondisi kode: `not self.position`
   * Akun wajib dalam keadaan tunai 100% (tidak sedang memegang posisi Long maupun Short). Algoritma tidak mengizinkan *averaging-down*, *pyramiding* (penambahan ukuran lot pada posisi yang berjalan), maupun *hedging* (membuka Long dan Short secara simultan).
2. **Masa Jeda Transaksi Belum Selesai Tidak Boleh Masuk (*Cooldown Enforcement*):**
   * Kondisi kode: `self.bars_since_exit >= self.cooldown_bars` (minimal 10 bar bursa = 10 jam perdagangan reguler atau ~1.5 hari bursa).
   * Jika posisi baru saja ditutup (baik terkena cut loss maupun profit lock), sistem dilarang membuka transaksi baru sebelum jeda 10 bar terlewati. Batasan ini menekan bias psikologis *revenge trading* dan mencegah kerugian berturut-turut pada fase konsolidasi/whipsaw.
3. **Periode Pemanasan Data Belum Mencukupi Tidak Boleh Masuk (*Warm-up Period*):**
   * Kondisi kode: `len(self.data) >= 16`
   * Algoritma menolak eksekusi jika bar per jam yang diproses masih di bawah periode EMA Mikro (16 bar) untuk menghindari sinyal palsu akibat kalkulasi rata-rata yang belum konvergen.
4. **Mode Perdagangan Belum Mengizinkan Tidak Boleh Masuk (*Execution Mode Filter*):**
   * Transaksi Beli (Long) hanya dievaluasi jika parameter mode diset ke `'long_only'` atau `'combined'`.
   * Transaksi Jual (Short) hanya dievaluasi jika parameter mode diset ke `'short_only'` atau `'combined'`.

---

### 2. Syarat Indikatornya Apa (Technical Indicator Entry Rules)

#### A. Sinyal Masuk Posisi Beli (LONG ENTRY)
Posisi Long dieksekusi (`self.buy(size=0.9999)`) pada harga penutupan bar saat ini (`Close[-1]`) jika dan hanya jika **seluruh 4 syarat teknikal berikut terpenuhi secara simultan**:
1. **Filter Tren Makro Harian (1D Bullish Regime):**
   * `self.macro_bullish[-1] == True`
   * Rumus: $	ext{EMA}_{21}(1	ext{D}) > 	ext{EMA}_{63}(1	ext{D}) 	imes (1 + 0.002)$
   * Garis tren cepat harian berada minimal 0.2% di atas garis tren lambat harian pada data kemarin (*yesterday's close*). Hal ini mengonfirmasi bahwa pasar berada dalam rezim *bullish* yang kuat secara struktural.
2. **Koreksi / Pullback Retest pada Timeframe 1H (Bar-2):**
   * `self.data.Low[-2] <= self.ema_micro[-2]`
   * Dua jam yang lalu, harga terendah (*Low*) sempat menyentuh atau menembus ke bawah garis rata-rata harga wajar mikro (EMA 16 jam). Ini membuktikan bahwa fase diskon/koreksi sehat sedang terjadi di dalam tren naik.
3. **Konfirmasi Pemantulan / Rebound pada Timeframe 1H (Bar-1):**
   * `self.data.Close[-1] > self.ema_micro[-1]`
   * Satu jam yang lalu, harga penutupan (*Close*) berhasil kembali ditutup di atas garis EMA Mikro 16 jam. Ini menandakan kembalinya minat beli dan penolakan terhadap harga murah.
4. **Konfirmasi Kemiringan Garis Mikro (*Slope Confirmation*):**
   * `self.ema_micro[-1] >= self.ema_micro[-2]`
   * Nilai EMA Mikro 16 jam saat ini tidak lebih rendah dari 1 jam sebelumnya. Memastikan bahwa momentum pelemahan mikro telah mereda dan garis moving average mulai mendatar atau melengkung ke atas.
5. **Penetapan Garis Risiko Awal (*Initial Stop Loss*):**
   * Seketika saat posisi Long dibuka, algoritma memasang garis pengaman awal sebesar 9.6 kali nilai ATR 12 jam di bawah harga beli:
     $$	ext{Trailing\_SL} = 	ext{EntryPrice} - (9.6 	imes 	ext{ATR}_{12})$$

#### B. Sinyal Masuk Posisi Jual (SHORT ENTRY)
Posisi Short dieksekusi (`self.sell(size=0.9999)`) pada harga penutupan bar saat ini (`Close[-1]`) jika dan hanya jika **seluruh 4 syarat teknikal berikut terpenuhi secara simultan**:
1. **Filter Tren Makro Harian (1D Bearish Regime):**
   * `self.macro_bearish[-1] == True`
   * Rumus: $	ext{EMA}_{21}(1	ext{D}) < 	ext{EMA}_{63}(1	ext{D}) 	imes (1 - 0.002)$
   * Garis tren cepat harian berada minimal 0.2% di bawah garis tren lambat harian pada data kemarin. Mengonfirmasi instrumen berada dalam rezim *bearish* yang dominan.
2. **Koreksi Naik / Pullback Retest Atas pada Timeframe 1H (Bar-2):**
   * `self.data.High[-2] >= self.ema_micro[-2]`
   * Dua jam yang lalu, harga tertinggi (*High*) sempat menguji atau menembus ke atas garis EMA Mikro 16 jam.
3. **Konfirmasi Penolakan Turun / Rejection pada Timeframe 1H (Bar-1):**
   * `self.data.Close[-1] < self.ema_micro[-1]`
   * Satu jam yang lalu, harga penutupan (*Close*) tertolak dan ditutup kembali di bawah garis EMA Mikro 16 jam. Menandakan dominasi tekanan jual jangka pendek.
4. **Konfirmasi Kemiringan Garis Mikro (*Slope Confirmation*):**
   * `self.ema_micro[-1] <= self.ema_micro[-2]`
   * Nilai EMA Mikro 16 jam saat ini lebih rendah atau sama dengan 1 jam sebelumnya, mengonfirmasi arah turunan momentum masih mengarah ke bawah.
5. **Penetapan Garis Risiko Awal (*Initial Stop Loss*):**
   * Seketika saat posisi Short dibuka, algoritma memasang garis pengaman awal sebesar 9.6 kali nilai ATR 12 jam di atas harga jual:
     $$	ext{Trailing\_SL} = 	ext{EntryPrice} + (9.6 	imes 	ext{ATR}_{12})$$

---

### 3. Mekanisme Pengunci Laba Berjalan (Trailing Stop / Ratchet Mechanism)
Ketika posisi sedang aktif berjalan, sistem mengevaluasi harga setiap penutupan candle 1 jam untuk memperketat proteksi keuntungan:
* **Ambang Batas Keaktifan (+2.3% Floating Profit):**
  * Selama posisi belum mencatat floating profit minimal **2.3%** (`Close[-1] < EntryPrice * 1.023` untuk Long, atau `Close[-1] > EntryPrice * 0.977` untuk Short), garis `Trailing_SL` **tetap diam di tempat** sebagai *Initial Cut Loss* guna memberi ruang napas dari noise intraday.
* **Pergeseran Ratchet untuk Posisi LONG:**
  * Begitu keuntungan belum terealisasi mencapai $\ge 2.3\%$, sistem menghitung kandidat batas baru: $	ext{new\_sl} = 	ext{Close}[-1] - (9.6 	imes 	ext{ATR})$.
  * Garis stop loss dinaikkan mengikuti titik tertinggi baru:
    $$	ext{Trailing\_SL} = \max(	ext{Trailing\_SL}, 	ext{new\_sl})$$
  * Garis stop loss Long tidak pernah diperbolehkan turun.
* **Pergeseran Ratchet untuk Posisi SHORT:**
  * Begitu keuntungan belum terealisasi mencapai $\ge 2.3\%$, sistem menghitung kandidat batas baru: $	ext{new\_sl} = 	ext{Close}[-1] + (9.6 	imes 	ext{ATR})$.
  * Garis stop loss diturunkan mengikuti titik terendah baru:
    $$	ext{Trailing\_SL} = \min(	ext{Trailing\_SL}, 	ext{new\_sl})$$
  * Garis stop loss Short tidak pernah diperbolehkan naik.

---

### 4. Keluar Ketika Apa (Kriteria Penutupan Posisi / Exit Rules)
Posisi yang sedang aktif **akan ditutup seketika (`self.position.close()`)** apabila memenuhi **salah satu dari 2 kondisi keluar mutlak berikut**:
1. **Kondisi Keluar 1: Penetrasi Trailing Stop / Initial Stop Loss (`Trailing Stop Hit`):**
   * **Posisi Long:** Harga penutupan per jam menembus ke bawah atau sama dengan garis stop loss (`Close[-1] <= self.trailing_sl`).
   * **Posisi Short:** Harga penutupan per jam menembus ke atas atau sama dengan garis stop loss (`Close[-1] >= self.trailing_sl`).
   * *Perilaku:* Jika posisi belum pernah mencapai keuntungan +2.3%, eksekusi ini berfungsi sebagai pemotong kerugian (*Cut Loss*). Jika posisi sudah berada dalam keuntungan besar, eksekusi ini berfungsi mengunci laba yang telah diraih (*Profit Lock*).
2. **Kondisi Keluar 2: Pembalikan Rezim Tren Makro Harian (`Macro Trend Reversal`):**
   * **Posisi Long:** Garis EMA 21 harian resmi berbalik memotong ke bawah garis EMA 63 harian (`not self.macro_bull_alive[-1]`).
   * **Posisi Short:** Garis EMA 21 harian resmi berbalik memotong ke atas garis EMA 63 harian (`not self.macro_bear_alive[-1]`).
   * *Perilaku:* Menutup posisi secara diskresioner karena landasan struktural tren besar (1D) yang menjadi prasyarat masuk telah gugur.

**Tindakan Pasca-Keluar (*Post-Exit Reset*):**
Seketika setelah transaksi ditutup oleh Kondisi Keluar 1 maupun 2, variabel `self.bars_since_exit` di-reset menjadi `0`. Ini mengunci sistem agar tidak dapat melakukan entry baru selama **10 bar perdagangan berikutnya** (*cooldown period*).

class SymmetricMTFStrategy(Strategy):
    """
    Strategi Kuantitatif Multi-Timeframe Simetris Penuh.
    Mendukung mode 'long_only', 'short_only', dan 'combined'.
    """
    mode = 'combined' # 'long_only', 'short_only', atau 'combined'
    atr_mult = GLOBAL_PARAMS['atr_mult']
    cooldown_bars = GLOBAL_PARAMS['cooldown_bars']
    min_profit_pct = GLOBAL_PARAMS['min_profit_pct']

    def init(self):
        # Indikator teknikal
        self.ema_micro = self.I(calc_ema_series, self.data.Close, GLOBAL_PARAMS['ema_micro'], overlay=True)
        self.atr = self.I(calc_atr_series, self.data.High, self.data.Low, self.data.Close, GLOBAL_PARAMS['atr_period'], overlay=False)
        self.macro_bullish = self.I(lambda: self.data.df['macro_bullish'].values, overlay=False)
        self.macro_bearish = self.I(lambda: self.data.df['macro_bearish'].values, overlay=False)
        self.macro_bull_alive = self.I(lambda: self.data.df['macro_bull_alive'].values, overlay=False)
        self.macro_bear_alive = self.I(lambda: self.data.df['macro_bear_alive'].values, overlay=False)

        # Variabel pelacak internal
        self.bars_since_exit = 999
        self.trailing_sl = 0.0
        self.entry_price = 0.0
        self.trade_records = []
        self.active_record = None

    def next(self):
        # Tahap 0: Validasi Warm-up Period
        if len(self.data) < GLOBAL_PARAMS['ema_micro']:
            return

        # Tahap 1: Evaluasi Entry jika Portofolio Kosong
        if not self.position:
            self.bars_since_exit += 1
            if self.bars_since_exit >= self.cooldown_bars:
                long_allowed = (self.mode in ['long_only', 'combined'])
                short_allowed = (self.mode in ['short_only', 'combined'])

                long_trigger = (
                    long_allowed and
                    self.macro_bullish[-1] and
                    (self.data.Low[-2] <= self.ema_micro[-2]) and
                    (self.data.Close[-1] > self.ema_micro[-1]) and
                    (self.ema_micro[-1] >= self.ema_micro[-2])
                )

                short_trigger = (
                    short_allowed and
                    self.macro_bearish[-1] and
                    (self.data.High[-2] >= self.ema_micro[-2]) and
                    (self.data.Close[-1] < self.ema_micro[-1]) and
                    (self.ema_micro[-1] <= self.ema_micro[-2])
                )

                if long_trigger:
                    self.buy(size=0.9999)
                    self.entry_price = self.data.Close[-1]
                    self.trailing_sl = self.entry_price - (self.atr_mult * self.atr[-1])
                    self.active_record = {
                        'EntryBar': len(self.data) - 1,
                        'EntryTime': self.data.index[-1],
                        'Mode': self.mode,
                        'Position_Type': 'LONG',
                        'ATR_At_Entry': self.atr[-1],
                        'Trailing_SL_Final': self.trailing_sl,
                        'Exit_Reason': None
                    }
                    return

                elif short_trigger:
                    self.sell(size=0.9999)
                    self.entry_price = self.data.Close[-1]
                    self.trailing_sl = self.entry_price + (self.atr_mult * self.atr[-1])
                    self.active_record = {
                        'EntryBar': len(self.data) - 1,
                        'EntryTime': self.data.index[-1],
                        'Mode': self.mode,
                        'Position_Type': 'SHORT',
                        'ATR_At_Entry': self.atr[-1],
                        'Trailing_SL_Final': self.trailing_sl,
                        'Exit_Reason': None
                    }
                    return

        # Tahap 2 & 3: Manajemen Posisi Berjalan (Ratchet SL & Exit Rules)
        else:
            if self.position.is_long:
                # Ratchet Trailing Stop
                is_in_profit = self.data.Close[-1] >= (self.entry_price * (1.0 + self.min_profit_pct))
                if is_in_profit:
                    new_sl = self.data.Close[-1] - (self.atr_mult * self.atr[-1])
                    self.trailing_sl = max(self.trailing_sl, new_sl)
                if self.active_record:
                    self.active_record['Trailing_SL_Final'] = self.trailing_sl

                # Exit 1: Trailing Stop Hit
                if self.data.Close[-1] <= self.trailing_sl:
                    if self.active_record:
                        self.active_record['Exit_Reason'] = 'Trailing Stop Hit'
                        self.trade_records.append(self.active_record.copy())
                        self.active_record = None
                    self.position.close()
                    self.bars_since_exit = 0
                    return

                # Exit 2: Macro Trend Reversal
                if not self.macro_bull_alive[-1]:
                    if self.active_record:
                        self.active_record['Exit_Reason'] = 'Macro Trend Reversal'
                        self.trade_records.append(self.active_record.copy())
                        self.active_record = None
                    self.position.close()
                    self.bars_since_exit = 0
                    return

            elif self.position.is_short:
                # Ratchet Trailing Stop
                is_in_profit = self.data.Close[-1] <= (self.entry_price * (1.0 - self.min_profit_pct))
                if is_in_profit:
                    new_sl = self.data.Close[-1] + (self.atr_mult * self.atr[-1])
                    self.trailing_sl = min(self.trailing_sl, new_sl)
                if self.active_record:
                    self.active_record['Trailing_SL_Final'] = self.trailing_sl

                # Exit 1: Trailing Stop Hit
                if self.data.Close[-1] >= self.trailing_sl:
                    if self.active_record:
                        self.active_record['Exit_Reason'] = 'Trailing Stop Hit'
                        self.trade_records.append(self.active_record.copy())
                        self.active_record = None
                    self.position.close()
                    self.bars_since_exit = 0
                    return

                # Exit 2: Macro Trend Reversal
                if not self.macro_bear_alive[-1]:
                    if self.active_record:
                        self.active_record['Exit_Reason'] = 'Macro Trend Reversal'
                        self.trade_records.append(self.active_record.copy())
                        self.active_record = None
                    self.position.close()
                    self.bars_since_exit = 0
                    return

print("Unified Strategy Class (SymmetricMTFStrategy) berhasil didefinisikan.")

def reconcile_trades(stats, ohlc_df, mode_label, fractional_unit=1e-4):

    trades = stats['_trades'].copy()
    if len(trades) == 0:
        return pd.DataFrame(columns=[
            'Trade_ID', 'Mode', 'Position_Type', 'Size', 'EntryTime', 'ExitTime',
            'Duration', 'Bars_Held', 'EntryPrice', 'ExitPrice', 'Trailing_SL_Final',
            'ATR_At_Entry', 'PnL_USD', 'Return_Pct', 'MAE_Pct', 'MFE_Pct',
            'Profit_Giveback_Pct', 'Exit_Reason'
        ])
    
    strat = stats['_strategy']
    records = list(strat.trade_records)
    
    # Tangani transaksi yang ditutup otomatis pada akhir rentang data oleh engine
    if strat.active_record is not None and len(records) < len(trades):
        rec_copy = strat.active_record.copy()
        rec_copy['Exit_Reason'] = 'End of Data'
        records.append(rec_copy)
        
    trade_rows = []
    
    for i in range(len(trades)):
        tr = trades.iloc[i]
        entry_bar = int(tr['EntryBar'])
        exit_bar = int(tr['ExitBar'])
        pos_size = float(tr['Size'])
        is_long = pos_size > 0
        pos_type = 'LONG' if is_long else 'SHORT'
        
        entry_px = float(tr['EntryPrice'])
        exit_px = float(tr['ExitPrice'])
        pnl = float(tr['PnL'])
        ret_pct = float(tr['ReturnPct']) * 100.0
        
        bars_held = exit_bar - entry_bar
        
        # Eliminasi Look-Behind Bias: Evaluasi MAE/MFE murni pada bar setelah entri Close
        if exit_bar > entry_bar:
            trade_slice = ohlc_df.iloc[entry_bar + 1 : exit_bar + 1]
        else:
            trade_slice = ohlc_df.iloc[entry_bar : entry_bar + 1]
        
        if is_long:
            lowest_price = float(trade_slice['Low'].min())
            mae_usd = min(0.0, lowest_price - entry_px)
            mae_pct = (mae_usd / entry_px) * 100.0
            
            highest_price = float(trade_slice['High'].max())
            mfe_usd = max(0.0, highest_price - entry_px)
            mfe_pct = (mfe_usd / entry_px) * 100.0
            
            if highest_price > entry_px:
                profit_giveback_pct = ((exit_px - highest_price) / highest_price) * 100.0
            else:
                profit_giveback_pct = 0.0
        else:
            highest_price = float(trade_slice['High'].max())
            mae_usd = min(0.0, entry_px - highest_price)
            mae_pct = (mae_usd / entry_px) * 100.0
            
            lowest_price = float(trade_slice['Low'].min())
            mfe_usd = max(0.0, entry_px - lowest_price)
            mfe_pct = (mfe_usd / entry_px) * 100.0
            
            if lowest_price < entry_px:
                profit_giveback_pct = ((lowest_price - exit_px) / lowest_price) * 100.0
            else:
                profit_giveback_pct = 0.0
                
        # Rekonsiliasi variabel internal pelacak
        if i < len(records):
            r = records[i]
            sl_final = r['Trailing_SL_Final'] / fractional_unit if fractional_unit > 0 else r['Trailing_SL_Final']
            atr_entry = r['ATR_At_Entry'] / fractional_unit if fractional_unit > 0 else r['ATR_At_Entry']
            exit_reason = r['Exit_Reason'] or ('End of Data' if exit_bar >= len(ohlc_df) - 1 else 'Trailing Stop Hit')
        else:
            sl_final = entry_px
            atr_entry = float(ohlc_df['ATR'].iloc[entry_bar])
            exit_reason = 'End of Data' if exit_bar >= len(ohlc_df) - 1 else 'Trailing Stop Hit'
            
        trade_rows.append({
            'Trade_ID': i + 1,
            'Mode': mode_label,
            'Position_Type': pos_type,
            'Size': round(pos_size, 4),
            'EntryTime': pd.to_datetime(tr['EntryTime']).strftime('%Y-%m-%d %H:%M'),
            'ExitTime': pd.to_datetime(tr['ExitTime']).strftime('%Y-%m-%d %H:%M'),
            'Duration': str(tr['Duration']),
            'Bars_Held': bars_held,
            'EntryPrice': round(entry_px, 2),
            'ExitPrice': round(exit_px, 2),
            'Trailing_SL_Final': round(sl_final, 2),
            'ATR_At_Entry': round(atr_entry, 4),
            'PnL_USD': round(pnl, 2),
            'Return_Pct': round(ret_pct, 2),
            'MAE_Pct': round(mae_pct, 2),
            'MFE_Pct': round(mfe_pct, 2),
            'Profit_Giveback_Pct': round(profit_giveback_pct, 2),
            'Exit_Reason': exit_reason
        })
        
    return pd.DataFrame(trade_rows)

Mengekstrak 10 metrik performa kuantitatif: Total Return (%), CAGR (%), Max Drawdown (%), Calmar Ratio, Sharpe Ratio (faktor tahunan 252 hari), Sortino Ratio (*Target Downside Deviation* = 0), Win Rate (%), Profit Factor, Total Trades, dan Exposure Time (%). Menjamin formula matematis yang identik antara strategi aktif dan *benchmark* pasif Buy & Hold.

# CELL 6: METRIC EXTRACTION & BUY & HOLD BENCHMARK
# =============================================================================
def extract_metrics(stats, df, is_bnh=False):
    """
    Ekstraksi metrik kuantitatif terstandarisasi dan apple-to-apple:
      - Total Return (%)
      - CAGR (%)
      - Max Drawdown (%)
      - Calmar Ratio
      - Sharpe Ratio
      - Sortino Ratio
      - Win Rate (%)
      - Profit Factor
      - Total Trades
      - Exposure Time (%)
    """
    dur_years = (df.index[-1] - df.index[0]).total_seconds() / (365.25 * 86400)

    if is_bnh:
        tot_ret = float(stats['Buy & Hold Return [%]'])
        cagr = ((1.0 + tot_ret / 100.0) ** (1.0 / dur_years) - 1.0) * 100.0 if tot_ret > -100.0 else -100.0

        equity_series = df['Close'] / df['Close'].iloc[0]
        mdd = float(((equity_series - equity_series.cummax()) / equity_series.cummax()).min() * 100.0)

        daily_ret = df['Close'].resample('D').last().dropna().pct_change().dropna()
        wr = np.nan
        pf = np.nan
        n_trades = 1
        exposure = 100.0
    else:
        tot_ret = float(stats['Return [%]'])
        cagr = float(stats['Return (Ann.) [%]'])
        mdd = float(stats['Max. Drawdown [%]'])

        # Gunakan kurva ekuitas riil strategi untuk resampling harian yang identik dengan BnH
        eq_curve = stats['_equity_curve']['Equity']
        daily_ret = eq_curve.resample('D').last().dropna().pct_change().dropna()

        wr = float(stats['Win Rate [%]']) if not pd.isna(stats['Win Rate [%]']) else 0.0
        pf = float(stats['Profit Factor']) if not pd.isna(stats['Profit Factor']) else 0.0
        n_trades = int(stats['# Trades'])
        exposure = float(stats['Exposure Time [%]']) if not pd.isna(stats['Exposure Time [%]']) else 0.0

    # Perhitungan Sharpe, Sortino (Target Downside Deviation = 0), dan Calmar yang seragam
    if len(daily_ret) > 1 and daily_ret.std() > 0:
        sharpe = float((daily_ret.mean() / daily_ret.std()) * np.sqrt(252))
    else:
        sharpe = 0.0

    downside_diff = np.minimum(0.0, daily_ret)
    downside_std = np.sqrt(np.mean(downside_diff ** 2))
    if len(daily_ret) > 1 and downside_std > 0:
        sortino = float((daily_ret.mean() / downside_std) * np.sqrt(252))
    else:
        sortino = 0.0

    calmar = float(cagr / abs(mdd)) if abs(mdd) > 0 else 0.0

    return pd.Series({
        'Total Return (%)': tot_ret,
        'CAGR (%)': cagr,
        'Max Drawdown (%)': mdd,
        'Calmar Ratio': calmar,
        'Sharpe Ratio': sharpe,
        'Sortino Ratio': sortino,
        'Win Rate (%)': wr,
        'Profit Factor': pf,
        'Total Trades': n_trades,
        'Exposure Time (%)': exposure
    })

print("Fungsi Ekstraksi Metrik Kuantitatif (extract_metrics) berhasil didefinisikan.")

# 1. KONFIGURASI UTAMA & JALUR BERKAS
# =============================================================================
BASE_DIR = Path(__file__).resolve().parent
DATA_DIR = BASE_DIR / "data_index"
if not DATA_DIR.exists():
    DATA_DIR = BASE_DIR.parent / "TugasAkhir" / "data_index"

OUTPUT_EXCEL = BASE_DIR / "Full_Grid_Search_Optimization.xlsx"
TARGET_INSTRUMENTS = ["SPY", "QQQ", "DJI"]

INITIAL_CAPITAL = 1000.0
FEE = 0.0003
EMA_MACRO_FAST = 12   # Mengikuti GLOBAL_PARAMS pada ta.ipynb
EMA_MACRO_SLOW = 26   # Mengikuti GLOBAL_PARAMS pada ta.ipynb
START_DATE = "2018-02-19"
END_DATE = "2024-05-10 23:59:59"


# =============================================================================
# 2. FUNGSI INDIKATOR & PRA-PEMROSESAN DATA MTF
# =============================================================================
def calc_ema_series(series, span):
    """Menghitung Exponential Moving Average (EMA) berbasis span."""
    return pd.Series(series).ewm(span=span, adjust=False).mean().values


def calc_atr_series(high, low, close, period=14):
    """Menghitung Average True Range (ATR) berbasis True Range EMA."""
    high_s = pd.Series(high)
    low_s = pd.Series(low)
    close_s = pd.Series(close)
    close_prev = close_s.shift(1)
    tr = pd.concat(
        [high_s - low_s, (high_s - close_prev).abs(), (low_s - close_prev).abs()],
        axis=1,
    ).max(axis=1)
    return pd.Series(tr).ewm(span=period, adjust=False).mean().values


def load_clean_data(file_path, is_1h=False):
    """Membaca data CSV/Parquet, filter akhir pekan, dan jam RTH 09:00-16:00 (1H)."""
    p = Path(file_path)
    if not p.exists():
        raise FileNotFoundError(f"Berkas data tidak ditemukan: {p}")

    if p.suffix.lower() == ".parquet":
        df = pd.read_parquet(p)
    else:
        with open(p, "r", encoding="utf-8") as f:
            sep = "\t" if "\t" in f.readline() else ","
        df = pd.read_csv(p, sep=sep)

    df.columns = [c.lower() for c in df.columns]
    if "datetime" in df.columns:
        df["datetime"] = pd.to_datetime(df["datetime"])
    elif "date" in df.columns:
        df["datetime"] = pd.to_datetime(df["date"])
        df.drop(columns=["date"], inplace=True, errors="ignore")
    elif "time" in df.columns:
        df["datetime"] = pd.to_datetime(df["time"])

    df.sort_values("datetime", inplace=True)
    df.set_index("datetime", inplace=True)
    df = df[df.index.dayofweek < 5]

    if is_1h:
        df = df[(df.index.hour >= 9) & (df.index.hour <= 16)]

    df["date"] = df.index.date
    return df


def get_asset_file_paths(asset):
    """Pemetaan berkas data 1D dan 1H per instrumen."""
    mapping = {
        "SPY": {"1d": DATA_DIR / "spy_US500_D1.parquet", "1h": DATA_DIR / "spy_US500_H1.parquet"},
        "QQQ": {"1d": DATA_DIR / "Nasdaq100_1d_data.csv", "1h": DATA_DIR / "Nasdaq100_1h_data.csv"},
        "DJI": {"1d": DATA_DIR / "Dow_Jones_1d_data.csv", "1h": DATA_DIR / "Dow_Jones_1h_data.csv"},
    }
    paths = mapping[asset]
    if not (paths["1d"].exists() and paths["1h"].exists()):
        raise FileNotFoundError(f"Data 1D/1H tidak lengkap untuk instrumen: {asset}")
    return paths


def prepare_optimization_data(asset):
    """Menyiapkan dataset gabungan 1H + EMA Makro 1D (shift 1 bar) bebas lookahead bias."""
    paths = get_asset_file_paths(asset)
    d1 = load_clean_data(paths["1d"], is_1h=False)
    h1 = load_clean_data(paths["1h"], is_1h=True)

    d1["ema_fast"] = pd.Series(calc_ema_series(d1["close"], EMA_MACRO_FAST), index=d1.index).shift(1)
    d1["ema_slow"] = pd.Series(calc_ema_series(d1["close"], EMA_MACRO_SLOW), index=d1.index).shift(1)
    d1["macro_bull_alive"] = d1["ema_fast"] >= d1["ema_slow"]

    h1 = h1[(h1.index >= pd.to_datetime(START_DATE)) & (h1.index <= pd.to_datetime(END_DATE))]

    d1_sub = d1[["date", "ema_fast", "ema_slow", "macro_bull_alive"]]
    df = pd.merge(h1.reset_index(), d1_sub, on="date", how="left")
    df.ffill(inplace=True)
    df.set_index("datetime", inplace=True)

    df.rename(
        columns={"open": "Open", "high": "High", "low": "Low", "close": "Close", "volume": "Volume"},
        inplace=True,
    )
    df.dropna(inplace=True)
    return df

# 3. STRATEGI & WORKER MULTIPROCESSING (WINDOWS-SAFE)
# =============================================================================
class OptMTFStrategy(Strategy):
    """Kelas strategi Long-Only untuk evaluasi kombinasi parameter Grid Search."""

    atr_mult = 6.0
    cooldown_bars = 7
    min_profit_pct = 0.010
    ema_micro_span = 33
    atr_period_span = 14

    def init(self):
        self.ema_micro = self.I(calc_ema_series, self.data.Close, self.ema_micro_span, overlay=True)
        self.atr = self.I(
            calc_atr_series, self.data.High, self.data.Low, self.data.Close, self.atr_period_span, overlay=False
        )
        self.macro_bullish = self.I(lambda: self.data.df["macro_bullish"].values, overlay=False)
        self.macro_bull_alive = self.I(lambda: self.data.df["macro_bull_alive"].values, overlay=False)
        self.bars_since_exit = 999
        self.trailing_sl = 0.0
        self.entry_price = 0.0

    def next(self):
        if len(self.data) < self.ema_micro_span:
            return

        if not self.position:
            self.bars_since_exit += 1
            if self.bars_since_exit >= self.cooldown_bars:
                long_trigger = (
                    self.macro_bullish[-1]
                    and (self.data.Low[-2] <= self.ema_micro[-2])
                    and (self.data.Close[-1] > self.ema_micro[-1])
                    and (self.ema_micro[-1] >= self.ema_micro[-2])
                )
                if long_trigger:
                    self.buy(size=0.9999)
                    self.entry_price = self.data.Close[-1]
                    self.trailing_sl = self.entry_price - (self.atr_mult * self.atr[-1])
        else:
            if self.data.Close[-1] >= (self.entry_price * (1.0 + self.min_profit_pct)):
                new_sl = self.data.Close[-1] - (self.atr_mult * self.atr[-1])
                self.trailing_sl = max(self.trailing_sl, new_sl)

            if self.data.Close[-1] <= self.trailing_sl or not self.macro_bull_alive[-1]:
                self.position.close()
                self.bars_since_exit = 0


PRELOADED_DATA = {}


def init_worker(preloaded_dict):
    """Inisialisasi dataset bersama pada masing-masing worker proses."""
    global PRELOADED_DATA
    PRELOADED_DATA = preloaded_dict


def evaluate_combination_task(combo_tuple):
    """Mengevaluasi satu kombinasi parameter secara agregat pada SPY, QQQ, dan DJI."""
    p_spread, p_ema_micro, p_atr_period, p_atr_mult, p_cooldown, p_min_profit = combo_tuple

    class TargetStrat(OptMTFStrategy):
        atr_mult = p_atr_mult
        cooldown_bars = p_cooldown
        min_profit_pct = p_min_profit
        ema_micro_span = p_ema_micro
        atr_period_span = p_atr_period

    asset_returns, asset_mdds, asset_trades = {}, {}, {}

    for asset in TARGET_INSTRUMENTS:
        base_df = PRELOADED_DATA[asset].copy(deep=False)
        base_df["macro_bullish"] = base_df["ema_fast"] > (base_df["ema_slow"] * (1.0 + p_spread))

        bt = FractionalBacktest(
            base_df,
            TargetStrat,
            cash=INITIAL_CAPITAL,
            commission=FEE,
            trade_on_close=True,
            finalize_trades=True,
            fractional_unit=1e-4,
        )
        stats = bt.run()

        asset_returns[asset] = float(stats["Return [%]"])
        asset_mdds[asset] = float(stats["Max. Drawdown [%]"])
        asset_trades[asset] = int(stats["# Trades"])

    avg_return = sum(asset_returns.values()) / len(TARGET_INSTRUMENTS)
    avg_mdd = sum(asset_mdds.values()) / len(TARGET_INSTRUMENTS)

    return {
        "macro_spread_threshold": p_spread,
        "ema_micro": p_ema_micro,
        "atr_period": p_atr_period,
        "atr_mult": p_atr_mult,
        "cooldown_bars": p_cooldown,
        "min_profit_pct": p_min_profit,
        "Avg_Total_Return": round(avg_return, 2),
        "Avg_Max_Drawdown": round(avg_mdd, 2),
        "SPY_Return (%)": round(asset_returns["SPY"], 2),
        "QQQ_Return (%)": round(asset_returns["QQQ"], 2),
        "DJI_Return (%)": round(asset_returns["DJI"], 2),
        "SPY_MaxDD (%)": round(asset_mdds["SPY"], 2),
        "QQQ_MaxDD (%)": round(asset_mdds["QQQ"], 2),
        "DJI_MaxDD (%)": round(asset_mdds["DJI"], 2),
        "SPY_Trades": asset_trades["SPY"],
        "QQQ_Trades": asset_trades["QQQ"],
        "DJI_Trades": asset_trades["DJI"],
    }

# 4. EKSPOR EXCEL TERFORMAT
# =============================================================================
def export_to_excel(df_optim_final, output_path):
    """Menyimpan seluruh peringkat dan Top 100 parameter ke Excel dengan styling standar."""
    with pd.ExcelWriter(output_path, engine="openpyxl") as writer:
        df_optim_final.to_excel(writer, sheet_name="Ranked_Universal_Parameters", index=False)
        df_optim_final.head(100).to_excel(writer, sheet_name="Top_100_Universal", index=False)

    wb = openpyxl.load_workbook(output_path)
    header_font = Font(name="Segoe UI", size=11, bold=True, color="FFFFFF")
    header_fill = PatternFill(start_color="1F4E78", end_color="1F4E78", fill_type="solid")

    for ws in wb.worksheets:
        ws.views.sheetView[0].showGridLines = True
        for cell in ws[1]:
            cell.font = header_font
            cell.fill = header_fill
            cell.alignment = Alignment(horizontal="center", vertical="center", wrap_text=True)

        for col in ws.columns:
            col_letter = get_column_letter(col[0].column)
            max_len = max(len(str(c.value or "")) for c in col)
            ws.column_dimensions[col_letter].width = max(max_len + 3, 13)

    wb.save(output_path)

# 5. EKSEKUSI UTAMA FULL GRID SEARCH (15.000 KOMBINASI)
# =============================================================================
def main():
    # Ruang pencarian Coarse Grid (5 x 5 x 5 x 6 x 5 x 4 = 15.000 kombinasi)
    macro_spread_threshold_range = np.round(np.arange(0.002, 0.022 + 0.0005, 0.005), 3).tolist()
    ema_micro_range = list(range(14, 47, 8))
    atr_period_range = list(range(7, 24, 4))
    atr_mult_range = np.round(np.arange(4.0, 10.0 + 0.05, 1.2), 1).tolist()
    cooldown_bars_range = list(range(3, 16, 3))
    min_profit_pct_range = np.round(np.arange(0.005, 0.023 + 0.0005, 0.006), 3).tolist()

    active_combinations = list(
        itertools.product(
            macro_spread_threshold_range,
            ema_micro_range,
            atr_period_range,
            atr_mult_range,
            cooldown_bars_range,
            min_profit_pct_range,
        )
    )
    assert len(active_combinations) == 15000, f"Jumlah kombinasi harus 15.000, ditemukan: {len(active_combinations)}"

    cpu_cores = os.cpu_count() or 4
    print("=" * 85)
    print("FULL GRID SEARCH PARAMETER OPTIMIZATION (EXHAUSTIVE 15.000 KOMBINASI)")
    print(f"Instrumen Target  : {TARGET_INSTRUMENTS} (Long-Only)")
    print(f"EMA Makro (1D)    : Fast={EMA_MACRO_FAST}, Slow={EMA_MACRO_SLOW}")
    print(f"Total Kombinasi   : {len(active_combinations):,} | CPU Workers: {cpu_cores} Cores")
    print(f"Berkas Output     : {OUTPUT_EXCEL}")
    print("=" * 85)

    # Pre-load data ke memori agar worker bebas disk I/O
    preloaded_data = {asset: prepare_optimization_data(asset) for asset in TARGET_INSTRUMENTS}
    for asset, df_a in preloaded_data.items():
        print(f"  * {asset}: {len(df_a):,} bar 1H siap dievaluasi")

    results_list = []
    start_time = time.time()

    with concurrent.futures.ProcessPoolExecutor(
        max_workers=cpu_cores,
        initializer=init_worker,
        initargs=(preloaded_data,),
    ) as executor:
        futures = [executor.submit(evaluate_combination_task, combo) for combo in active_combinations]
        for fut in tqdm(
            concurrent.futures.as_completed(futures),
            total=len(futures),
            desc="Full Grid Search",
            unit="combo",
        ):
            results_list.append(fut.result())

    elapsed_min = (time.time() - start_time) / 60.0
    print(f"\nOptimasi selesai dalam {elapsed_min:.2f} menit ({len(results_list):,} kombinasi).")

    df_optim = pd.DataFrame(results_list)
    df_optim.sort_values(by=["Avg_Total_Return", "Avg_Max_Drawdown"], ascending=[False, False], inplace=True)
    df_optim.reset_index(drop=True, inplace=True)

    final_columns = [
        "Avg_Total_Return", "Avg_Max_Drawdown",
        "SPY_Return (%)", "QQQ_Return (%)", "DJI_Return (%)",
        "SPY_MaxDD (%)", "QQQ_MaxDD (%)", "DJI_MaxDD (%)",
        "macro_spread_threshold", "ema_micro", "atr_period", "atr_mult",
        "cooldown_bars", "min_profit_pct",
        "SPY_Trades", "QQQ_Trades", "DJI_Trades",
    ]
    df_optim_final = df_optim[final_columns]
    export_to_excel(df_optim_final, OUTPUT_EXCEL)

    print(f"\nHasil tersimpan: {OUTPUT_EXCEL}")
    print("\n>>> TOP 10 PARAMETER UNIVERSAL TERBAIK FULL GRID SEARCH:")
    pd.set_option("display.max_columns", None)
    pd.set_option("display.width", 1000)
    print(df_optim_final.head(10).to_string(index=False))


if __name__ == "__main__":
    main()
